
def D.dVal(v: D.Val): Unit = v match
  case D.VSym(s) =>
    Console.print("\"")
    Console.print(Sym.str(s))
    Console.print("\"")
  case D.VInt(n) => Console.printI64(n)

def D.dArg(a: D.Arg): Unit = a match
  case D.AVar(n) =>
    Console.print("v")
    Console.printI64(n)
  case D.AVal(v) => D.dVal(v)

def D.dArgs(xs: Coll.List[D.Arg], first: Bool): Unit = xs match
  case Coll.Nil => ()
  case Coll.Cons(x, rest) =>
    if first then () else Console.print(", ")
    D.dArg(x)
    D.dArgs(rest, false)

def D.dAtom(a: D.Atom): Unit =
  val D.Atom(r, args) = a
  Console.print(Sym.str(r))
  Console.print("(")
  D.dArgs(args, true)
  Console.print(")")

def D.dLits(xs: Coll.List[D.Lit], first: Bool): Unit = xs match
  case Coll.Nil => ()
  case Coll.Cons(x, rest) =>
    if first then () else Console.print(", ")
    x match
      case D.Pos(a) => D.dAtom(a)
      case D.Neg(a) =>
        Console.print("!")
        D.dAtom(a)
    D.dLits(rest, false)

def D.dRules(xs: Coll.List[D.Rule]): Unit = xs match
  case Coll.Nil => ()
  case Coll.Cons(D.Rule(h, b), rest) =>
    Console.print("RULE\t")
    D.dAtom(h)
    Console.print(" :- ")
    D.dLits(b, true)
    Console.print(".\n")
    D.dRules(rest)

def D.dTag(v: D.Val): Unit = v match
  case D.VSym(s) =>
    Console.print("\tS:")
    Console.print(Sym.str(s))
  case D.VInt(n) =>
    Console.print("\tI:")
    Console.printI64(n)

def D.dVals(xs: Coll.List[D.Val]): Unit = xs match
  case Coll.Nil => ()
  case Coll.Cons(x, rest) =>
    D.dTag(x)
    D.dVals(rest)

def D.dFacts(xs: Coll.List[D.Fact]): Unit = xs match
  case Coll.Nil => ()
  case Coll.Cons(D.Fact(r, vs), rest) =>
    Console.print("FACT\t")
    Console.print(Sym.str(r))
    D.dVals(vs)
    Console.print("\n")
    D.dFacts(rest)

def D.dump(rules: Coll.List[D.Rule], facts: Coll.List[D.Fact]): D.Result =
  Console.print("###SOLVE 1\n")
  D.dRules(rules)
  D.dFacts(facts)
  Console.print("###END\n")
  D.solve(rules, facts)
